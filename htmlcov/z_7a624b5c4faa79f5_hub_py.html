1"""A thin client for the Harbor hub (``org/name@ref`` datasets and tasks). 

2 

3The hub has no dedicated HTTP API: Harbor's own client talks to a Supabase 

4project through PostgREST table queries, one RPC, and a storage bucket, all 

5readable anonymously with the publishable key. This module does the same 

6with ``httpx`` so inspect_harbor does not depend on the ``harbor`` package. 

7 

8Verified against the live backend on 2026-09-21. Table and column names are 

9Harbor internals and may change without notice. The nightly registry update 

10resolves every hub dataset through this client, so a change to the dataset 

11or task-listing queries fails that job; the task-version RPC and archive 

12download are only exercised by evals and the opt-in parity test in 

13``tests/manual``. 

14"""

367def classify_ref(ref: str | None) -> tuple[RefType, str]: 

368 """Split a ref into its type and bare value. 

369 

370 ``None``, ``""`` and ``"latest"`` are the ``latest`` tag; digits are a 

371 revision number; ``sha256:<hex>`` is a digest (returned lower-cased and 

372 without the prefix); anything else is a tag. 

373 """ 

374 if not ref or ref == "latest": 

375 return RefType.TAG, "latest" 

376 if ref.isdigit(): 

377 return RefType.REVISION, ref 

378 if ref.startswith("sha256:"): 

379 return RefType.DIGEST, ref.removeprefix("sha256:").lower() 

380 return RefType.TAG, ref

383def parse_package_ref(slug: str) -> PackageRef: 

384 """Parse ``org/name`` or ``org/name@ref`` (ref defaults to ``latest``). 

385 

386 Raises: 

387 ValueError: When the name is not in ``org/name`` form. 

388 """ 

389 name, _, ref = slug.partition("@") 

390 if not re.match(ORG_NAME_PATTERN, name) or ".." in name: 

391 raise ValueError( 

392 f"Package name must be in 'org/name' format (got {slug!r}); allowed " 

393 "characters are letters, digits, '.', '_' and '-'." 

394 ) 

395 org, short = name.split("/", 1) 

396 return PackageRef(org=org, name=short, ref=ref or "latest")

423async def download_hub_tasks( 

424 task_refs: list[HubTaskRef], 

425 overwrite: bool = False, 

426 client: HubClient | None = None, 

427 max_concurrency: int = 100, 

428) -> list[Path]: 

429 """Download pinned hub tasks into the cache; returns paths in input order. 

430 

431 Tasks are content-addressed, so a complete cached directory is reused 

432 unless ``overwrite`` is set. Concurrent downloads of the same task, from 

433 this or another process, are safe: each extracts into its own staging 

434 directory and the first to finish wins. 

435 

436 ``max_concurrency`` defaults to 100, which is httpx's connection-pool 

437 ceiling and what Harbor uses against the same backend. Measured on 100 

438 aider tasks: 8 in flight took 11.5 s, 32 took 3.7 s, 100 took 2.2 s with 

439 no throttling, so higher values could not help and lower ones cost time 

440 on large datasets. 

441 """ 

442 targets = {ref: hub_task_dir(ref) for ref in task_refs} 

443 missing = [ 

444 ref for ref, target in targets.items() if overwrite or not is_cached(target) 

445 ] 

446 if missing: 

447 own_client = client is None 

448 client = client or HubClient() 

449 semaphore = asyncio.Semaphore(max_concurrency) 

450 try: 

451 async with asyncio.TaskGroup() as tg: 

452 for ref in missing: 

453 tg.create_task( 

454 _download_one(client, ref, targets[ref], overwrite, semaphore) 

455 ) 

456 except* Exception as group: 

457 first, *others = group.exceptions 

458 for other in others: 458 ↛ 459line 458 didn't jump to line 459 because the loop on line 458 never started

459 logger.warning("Additional hub download failure: %r", other) 

460 raise first from None 

461 finally: 

462 if own_client: 462 ↛ 463line 462 didn't jump to line 463 because the condition on line 462 was never true

463 await client.aclose() 

464 return [targets[ref] for ref in task_refs]

483def _extract_archive(archive: Path, target: Path, overwrite: bool) -> bool: 

484 """Extract into a private staging dir, then swap it into place. 

485 

486 Returns ``False`` when another writer completed ``target`` first and this 

487 extraction was discarded (only when not overwriting). 

488 

489 Raises: 

490 ValueError: When the archive does not contain a ``task.toml`` at its 

491 root, so a bad archive never becomes a cached "task". 

492 """ 

493 target.parent.mkdir(parents=True, exist_ok=True) 

494 staging = Path(tempfile.mkdtemp(dir=target.parent, prefix=f".{target.name}.")) 

495 staged = staging / "task" 

496 try: 

497 staged.mkdir() 

498 with tarfile.open(archive, "r:gz") as tar: 

499 tar.extractall(path=staged, filter="data") 

500 if not (staged / "task.toml").is_file(): 

501 raise ValueError( 

502 f"Hub archive for {target.name} has no task.toml at its root" 

503 ) 

504 if target.exists(): 

505 if not overwrite and is_cached(target): 

506 return False 

507 old = staging / "old" 

508 target.rename(old) 

509 try: 

510 staged.rename(target) 

511 except OSError: 

512 # Lost a race: someone else renamed a complete task into place. 

513 if is_cached(target): 

514 return False 

515 raise 

516 return True 

517 finally: 

518 shutil.rmtree(staging, ignore_errors=True)
